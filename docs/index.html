04question → Which regions grew fastest last quarter?
-- generated SQL
SELECT region, SUM(total) AS revenue_growth
FROM orders JOIN customers USING (id)
WHERE created_at >= '2026-04-01'
GROUP BY region ORDER BY revenue_growth DESC;
